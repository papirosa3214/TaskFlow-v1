import XCTest

final class TaskFamilyNavigationUITests: XCTestCase {
    override func setUp() {
        super.setUp()
        setSaveFailure(false)
        resetCard("11e72477-6241-4506-84be-fac3a5a2750d", fields: ["title": "Выпустить авторизацию"])
        resetCard("6ca26c5f-a1b8-4607-89e1-2d97a15ad7ac", fields: ["description": "Подготовить методы входа, выхода и восстановления доступа.\n\nОтветы об ошибках согласовать с экраном входа."])
    }
    private func resetCard(_ id: String, fields: [String: String]) {
        var request = URLRequest(url: URL(string: "http://127.0.0.1:3307/api/tasks/\(id)")!)
        request.httpMethod = "PATCH"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue("Bearer " + ProcessInfo.processInfo.environment["TF_CARD_DEMO_TOKEN"]!, forHTTPHeaderField: "Authorization")
        request.httpBody = try! JSONSerialization.data(withJSONObject: fields)
        let done = expectation(description: "Reset isolated fixture")
        URLSession.shared.dataTask(with: request) { _, response, error in
            XCTAssertNil(error)
            XCTAssertEqual((response as? HTTPURLResponse)?.statusCode, 200)
            done.fulfill()
        }.resume()
        wait(for: [done], timeout: 5)
    }
    func testFamilyAndDeliberateEditing() {
        let app = XCUIApplication()
        app.launchEnvironment["TASKFLOW_LINEAR_CARD_DEMO"] = "1"
        app.launchEnvironment["TASKFLOW_DEBUG_TOKEN"] = ProcessInfo.processInfo.environment["TF_CARD_DEMO_TOKEN"]!
        let root = "11e72477-6241-4506-84be-fac3a5a2750d"
        let child = "6ca26c5f-a1b8-4607-89e1-2d97a15ad7ac"
        app.launchEnvironment["TASKFLOW_DEBUG_ROUTE"] = "tasksheet:\(root)"
        app.launch()
        let arm = app.buttons["task.arm.title"]
        XCTAssertTrue(arm.waitForExistence(timeout: 30))
        guard expandSheet(app) else { return }
        let childTab = app.buttons["task.family.\(child)"]
        XCTAssertTrue(childTab.waitForExistence(timeout: 15))
        capture(app, "parent-bookmarks")
        arm.tap()
        let edit = app.buttons["task.edit.title"]
        XCTAssertTrue(edit.waitForExistence(timeout: 5))
        XCTAssertFalse(app.keyboards.firstMatch.exists)
        capture(app, "pencil-without-keyboard")
        edit.tap()
        XCTAssertEqual(edit.label, "Сохранить изменения")
        let title = app.textViews.firstMatch
        XCTAssertTrue(app.keyboards.firstMatch.waitForExistence(timeout: 5))
        title.typeText(" — проверка перехода")
        childTab.press(forDuration: 0.1, thenDragTo: app.descendants(matching: .any)["task.family.active.\(root)"])
        expectation(for: NSPredicate(format: "label CONTAINS %@", "Реализовать API"), evaluatedWith: app.buttons["task.arm.title"])
        waitForExpectations(timeout: 15)
        XCTAssertTrue(app.buttons["task.family.\(root)"].waitForExistence(timeout: 10))
        XCTAssertTrue(app.buttons["task.arm.title"].waitForExistence(timeout: 10))
        capture(app, "child-bookmarks")
        app.buttons["task.family.\(root)"].tap()
        let updated = NSPredicate(format: "label CONTAINS %@", "проверка перехода")
        expectation(for: updated, evaluatedWith: app.buttons["task.arm.title"])
        waitForExpectations(timeout: 15)
        app.buttons["task.arm.title"].tap()
        app.buttons["task.edit.title"].tap()
        app.buttons["task.edit.title"].tap()
        XCTAssertTrue(app.buttons["task.arm.title"].waitForExistence(timeout: 10))
        XCTAssertFalse(app.buttons["task.edit.title"].exists)
        capture(app, "saved-parent")
    }
    func testFailedSaveKeepsCurrentCard() {
        let app = XCUIApplication()
        app.launchEnvironment["TASKFLOW_LINEAR_CARD_DEMO"] = "1"
        app.launchEnvironment["TASKFLOW_DEBUG_TOKEN"] = ProcessInfo.processInfo.environment["TF_CARD_DEMO_TOKEN"]!
        let root = "11e72477-6241-4506-84be-fac3a5a2750d"
        let child = "6ca26c5f-a1b8-4607-89e1-2d97a15ad7ac"
        app.launchEnvironment["TASKFLOW_DEBUG_ROUTE"] = "tasksheet:\(root)"
        app.launch()
        XCTAssertTrue(app.buttons["task.arm.title"].waitForExistence(timeout: 30))
        guard expandSheet(app) else { return }
        app.buttons["task.arm.title"].tap()
        app.buttons["task.edit.title"].tap()
        app.textViews.firstMatch.typeText(" — ошибка сохранения")
        setSaveFailure(true)
        defer { setSaveFailure(false) }
        app.buttons["task.family.\(child)"].tap()
        XCTAssertTrue(app.staticTexts.containing(NSPredicate(format: "label CONTAINS %@", "Тестовая ошибка сохранения")).firstMatch.waitForExistence(timeout: 10))
        XCTAssertTrue(app.buttons["task.edit.title"].exists)
        XCTAssertTrue(app.textViews.firstMatch.value as? String != nil)
        XCTAssertTrue((app.textViews.firstMatch.value as? String)?.contains("ошибка сохранения") == true)
        capture(app, "failed-save-keeps-card")
        setSaveFailure(false)
        app.buttons["task.family.\(child)"].tap()
        expectation(for: NSPredicate(format: "label CONTAINS %@", "Реализовать API"), evaluatedWith: app.buttons["task.arm.title"])
        waitForExpectations(timeout: 15)
    }

    func testDirectChildEntryAndDescriptionEditing() {
        let app = XCUIApplication()
        app.launchEnvironment["TASKFLOW_LINEAR_CARD_DEMO"] = "1"
        app.launchEnvironment["TASKFLOW_DEBUG_TOKEN"] = ProcessInfo.processInfo.environment["TF_CARD_DEMO_TOKEN"]!
        let child = "6ca26c5f-a1b8-4607-89e1-2d97a15ad7ac"
        let root = "11e72477-6241-4506-84be-fac3a5a2750d"
        app.launchEnvironment["TASKFLOW_DEBUG_ROUTE"] = "tasksheet:\(child)"
        app.launch()
        XCTAssertTrue(app.buttons["task.arm.title"].waitForExistence(timeout: 30))
        guard expandSheet(app) else { return }
        XCTAssertTrue(app.buttons["task.family.\(root)"].exists)
        XCTAssertEqual(app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH %@", "task.family.")).count, 5)
        app.buttons["Описание"].tap()
        let arm = app.buttons["task.arm.description"]
        XCTAssertTrue(arm.waitForExistence(timeout: 5))
        arm.tap()
        XCTAssertFalse(app.keyboards.firstMatch.exists)
        app.buttons["task.edit.description"].tap()
        XCTAssertTrue(app.keyboards.firstMatch.waitForExistence(timeout: 5))
        let description = app.textViews.containing(NSPredicate(format: "value BEGINSWITH %@", "Подготовить методы")).firstMatch
        description.typeText(" Проверка описания.")
        app.buttons["task.edit.description"].tap()
        XCTAssertTrue(arm.waitForExistence(timeout: 10))
        XCTAssertTrue(arm.label.contains("Проверка описания"))
        XCTAssertFalse(app.buttons["task.edit.description"].exists)
        capture(app, "child-description-saved")
        app.buttons["task.family.\(root)"].tap()
        expectation(for: NSPredicate(format: "label CONTAINS %@", "Выпустить авторизацию"), evaluatedWith: app.buttons["task.arm.title"])
        waitForExpectations(timeout: 15)
    }

    func testCollapseHidesBookmarksAndVerticalEdgeBrowseDoesNotNavigate() {
        let app = XCUIApplication()
        app.launchEnvironment["TASKFLOW_LINEAR_CARD_DEMO"] = "1"
        app.launchEnvironment["TASKFLOW_DEBUG_TOKEN"] = ProcessInfo.processInfo.environment["TF_CARD_DEMO_TOKEN"]!
        let root = "11e72477-6241-4506-84be-fac3a5a2750d"
        let child = "6ca26c5f-a1b8-4607-89e1-2d97a15ad7ac"
        app.launchEnvironment["TASKFLOW_DEBUG_ROUTE"] = "tasksheet:\(root)"
        app.launch()
        XCTAssertTrue(app.buttons["task.arm.title"].waitForExistence(timeout: 30))
        guard expandSheet(app) else { return }
        app.buttons["task.family.\(child)"].swipeDown()
        XCTAssertTrue(app.buttons["task.arm.title"].label.contains("Выпустить авторизацию"))
        XCTAssertTrue(app.descendants(matching: .any)["task.family.active.\(root)"].exists)
        app.buttons["Sheet Grabber"].swipeDown()
        expectation(for: NSPredicate(format: "exists == false"), evaluatedWith: app.buttons["task.family.\(child)"])
        waitForExpectations(timeout: 10)
        XCTAssertTrue(app.buttons["task.arm.title"].exists)
        capture(app, "collapsed-again-no-bookmarks")
    }

    func testRecordEdgePull() {
        let app = XCUIApplication()
        app.launchEnvironment["TASKFLOW_LINEAR_CARD_DEMO"] = "1"
        app.launchEnvironment["TASKFLOW_DEBUG_TOKEN"] = ProcessInfo.processInfo.environment["TF_CARD_DEMO_TOKEN"]!
        let root = "11e72477-6241-4506-84be-fac3a5a2750d"
        let child = "6ca26c5f-a1b8-4607-89e1-2d97a15ad7ac"
        app.launchEnvironment["TASKFLOW_DEBUG_ROUTE"] = "tasksheet:\(root)"
        app.launch()
        XCTAssertTrue(app.buttons["task.arm.title"].waitForExistence(timeout: 30))
        guard expandSheet(app) else { return }
        app.buttons["task.family.\(child)"].press(forDuration: 0.2,
            thenDragTo: app.descendants(matching: .any)["task.family.active.\(root)"])
        expectation(for: NSPredicate(format: "label CONTAINS %@", "Реализовать API"), evaluatedWith: app.buttons["task.arm.title"])
        waitForExpectations(timeout: 15)
        capture(app, "child-after-edge-pull")
        app.buttons["task.family.\(root)"].press(forDuration: 0.2,
            thenDragTo: app.descendants(matching: .any)["task.family.active.\(child)"])
        expectation(for: NSPredicate(format: "label CONTAINS %@", "Выпустить авторизацию"), evaluatedWith: app.buttons["task.arm.title"])
        waitForExpectations(timeout: 15)
        capture(app, "parent-after-edge-pull")
    }

    private func expandSheet(_ app: XCUIApplication) -> Bool {
        XCTAssertFalse(app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH %@", "task.family.")).firstMatch.exists)
        capture(app, "half-sheet-no-bookmarks")
        let handle = app.buttons["Sheet Grabber"]
        XCTAssertTrue(handle.exists)
        handle.swipeUp()
        let ready = app.buttons.matching(NSPredicate(format: "identifier BEGINSWITH %@", "task.family.")).firstMatch.waitForExistence(timeout: 15)
        XCTAssertTrue(ready)
        return ready
    }

    private func setSaveFailure(_ fail: Bool) {
        var request = URLRequest(url: URL(string: "http://127.0.0.1:3307/__test/fail-save")!)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try! JSONSerialization.data(withJSONObject: ["fail": fail])
        let done = expectation(description: "Test server save switch")
        URLSession.shared.dataTask(with: request) { _, response, error in
            XCTAssertNil(error)
            XCTAssertEqual((response as? HTTPURLResponse)?.statusCode, 200)
            done.fulfill()
        }.resume()
        wait(for: [done], timeout: 5)
    }

    private func capture(_ app: XCUIApplication, _ name: String) {
        let attachment = XCTAttachment(screenshot: app.screenshot())
        attachment.name = name
        attachment.lifetime = .keepAlways
        add(attachment)
    }
}
